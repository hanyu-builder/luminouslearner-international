ctx.lineTo(ZONES.AERATION.x + 20, PIPE_BOTTOM); ctx.lineTo(ZONES.SECONDARY.x + ZONES.SECONDARY.w/2, PIPE_BOTTOM); ctx.lineTo(ZONES.SECONDARY.x + ZONES.SECONDARY.w/2, TANK_BOTTOM); ctx.stroke(); Object.values(ZONES).forEach(zone => { if (zone.name) { ctx.strokeStyle = '#64748B'; Secondary Clarifier (Biomass Settling) else if (p.x > ZONES.SECONDARY.x && p.x < ZONES.SECONDARY.x + ZONES.SECONDARY.w) { if (p.type === TYPES.BACTERIA) { p.vy = 0.8; // Floc settles out p.vx *= 0.3; } else if (p.type === TYPES.ORGANIC || p.type === TYPES.SOLID) { // Unprocessed waste mostly passes through, some slight settling p.vy = 0.2; } } // 4. p.y = TANK_BOTTOM + 5; } else { p.y = TANK_BOTTOM; p.vy = 0; } } } // RAS Logic (Return Activated Sludge Pump) else if (p.state === 'ras') { p.y = PIPE_BOTTOM - 10 + (Math.random() * 5); p.x -= 3; // Pump fast backward if (p.x < ZONES.AERATION.x + 30) { // Shoot back up into aeration basin p.state = 'flow';
